\section{The Syllabus of the International Physics Olympiad}

\subsection*{1. Mechanics}

\begin{enumerate}[label=\arabic*.,leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item \textbf{Kinematics.} Motion of a point mass; position, displacement, velocity and acceleration;
motion with uniform velocity and uniformly accelerated motion; motion under gravity, projectile
motion; motion in a circle with constant speed; angular velocity and angular acceleration.
\item \textbf{Dynamics.} Newton's laws of motion; inertial frames of reference; momentum and its
conservation; centre of mass; variable mass (rocket motion); friction; elastic and inelastic
collisions.
\item \textbf{Work and energy.} Work done by a force; kinetic energy; conservative forces and
potential energy; conservation of mechanical energy; power; efficiency.
\item \textbf{Rotational motion.} Rotation of a rigid body about a fixed axis; torque; moment of
inertia; angular momentum and its conservation; conditions of equilibrium of a rigid body.
\item \textbf{Gravitation.} Newton's law of gravitation; gravitational field and potential; Kepler's
laws; motion of planets and satellites; escape velocity.
\item \textbf{Hydrostatics and hydrodynamics.} Pressure in a fluid; Pascal's law; Archimedes'
principle and floating; steady flow of an ideal fluid; Bernoulli's equation; viscosity; Stokes'
law; Reynolds number.
\item \textbf{Elasticity.} Hooke's law; Young's modulus; shear modulus; elastic energy.
\item \textbf{Oscillations.} Simple harmonic motion; pendulum and spring oscillator; composition of
oscillations; damped and forced oscillations; resonance.
\item \textbf{Waves.} Travelling waves; longitudinal and transverse waves; wave equation;
superposition; reflection, refraction and interference of waves; standing waves; sound waves and
the Doppler effect.
\end{enumerate}

\subsection*{2. Thermodynamics and Molecular Physics}

\begin{enumerate}[label=\arabic*.,leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item Ideal gas; equation of state; kinetic theory of gases; pressure and temperature; equipartition
of energy; mean free path; diffusion.
\item First law of thermodynamics; internal energy; work and heat; specific heats of gases;
isothermal, isobaric, isochoric and adiabatic processes.
\item Second law of thermodynamics; reversible and irreversible processes; entropy; Carnot cycle;
efficiency of heat engines; refrigerators and heat pumps.
\item Real gases; van der Waals equation; phase transitions; saturated vapour; humidity.
\item Heat transfer: conduction, convection and radiation; thermal expansion; calorimetry.
\end{enumerate}

\subsection*{3. Electromagnetism}

\begin{enumerate}[label=\arabic*.,leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item \textbf{Electrostatics.} Electric charge; Coulomb's law; electric field; Gauss's law; electric
potential; conductors and dielectrics; capacitance and capacitors; energy of the electric field.
\item \textbf{Direct-current circuits.} Electric current; Ohm's law; resistance and resistivity;
series and parallel connections; Kirchhoff's rules; work and power of a current; electrolysis.
\item \textbf{Magnetic field.} Magnetic field of currents; Biot--Savart law; Amp\`ere's circuital
law; force on a current and on a moving charge; magnetic properties of matter.
\item \textbf{Electromagnetic induction.} Faraday's law; Lenz's law; self-inductance and mutual
inductance; energy of the magnetic field.
\item \textbf{Alternating currents.} RLC circuits; impedance; resonance; power; transformers.
\item \textbf{Electromagnetic waves.} Production and properties; the electromagnetic spectrum;
energy and momentum of electromagnetic radiation.
\end{enumerate}

\subsection*{4. Optics}

\begin{enumerate}[label=\arabic*.,leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item \textbf{Geometrical optics.} Laws of reflection and refraction; total internal reflection;
plane mirrors and spherical mirrors; lenses; image formation; optical instruments; the eye.
\item \textbf{Wave optics.} Coherence; interference of light; Young's experiment; thin films;
diffraction; diffraction gratings; resolving power; polarisation; dispersion of light.
\end{enumerate}

\subsection*{5. Modern Physics}

\begin{enumerate}[label=\arabic*.,leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item Atomic structure; spectra of atoms; the Bohr model of the hydrogen atom; energy levels;
X-rays; the Moseley law.
\item Photoelectric effect; photons; the Compton effect; de Broglie waves; the uncertainty
principle.
\item Nuclear physics: the atomic nucleus; radioactivity; the law of radioactive decay;
half-life; nuclear reactions; binding energy; fission and fusion; interaction of radiation with
matter.
\item The special theory of relativity: the principle of relativity; Lorentz transformations;
time dilation and length contraction; relativistic momentum and energy; the equivalence of mass
and energy.
\end{enumerate}

\subsection*{6. Experimental Skills}

\begin{enumerate}[label=\arabic*.,leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item Use of measuring instruments: vernier calipers, micrometer screw gauge, balances,
thermometers, ammeters and voltmeters, galvanometers, oscilloscopes, optical instruments.
\item Measurement of physical quantities and estimation of errors; systematic and random errors;
error propagation; significant figures.
\item Presentation of results: tables, graphs, linearisation of relations, least-squares fitting.
\item Design of simple experiments and critical discussion of the results.
\end{enumerate}
