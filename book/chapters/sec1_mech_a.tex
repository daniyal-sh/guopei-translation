\section{Mechanics}

\begin{problem}{1.01}
A small ball is thrown from the ground over a high wall of height $h$ situated at a horizontal distance $d$. Find the minimum initial speed $v_{0\min}$ required and the corresponding angle of projection $\theta$.
\end{problem}

\begin{problem}{1.02}
A small ball is thrown from the ground over two parallel walls, each of height $h$ and separated by a distance $d$. Find the minimum initial speed required, the corresponding angle of projection, and the distance from the point of projection to the nearer wall.
\end{problem}

\begin{problem}{1.04}
A lamp hangs at a distance $h_1$ below the ceiling, and the lamp is at a height $h_2$ above the floor. The bulb suddenly bursts into many fragments, all of which fly off in every direction with the same initial speed $v_0$. The collisions of the fragments with the ceiling are elastic; the fragments do not rebound from the floor and do not strike the walls. Find the radius of the region in which the fragments land.
\end{problem}

\begin{problem}{1.05}
A person stands at a distance $d$ from a high wall and plays a shooting game. Each bullet leaves the gun with the same initial speed $v_0$.
\begin{enumerate}[label=(\arabic*),leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item To hit the same horizontal line at height $h$ on the wall, find the locus of the aiming points on the wall.
\item If the points of impact on the wall are to lie above this horizontal line, find the range of the aiming points.
\end{enumerate}
\end{problem}

\begin{problem}{1.06}
An aircraft flies horizontally at a height $h$ above the ground with speed $v_0$ and drops a bomb on a ground target.
\begin{enumerate}[label=(\arabic*),leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item At what horizontal distance $L$ from the target must the bomb be released in order that it hit the target?

\end{problem}

\begin{problem}{1.07}
A circle of radius $R$ moves uniformly in the plane of the paper with speed $v_1$ to the right, while a straight line $MN$ moves uniformly upward with speed $v_2$. The line meets the circle at the point $P$. When the angle between $OP$, where $O$ is the centre of the circle, and $MN$ is $\theta$, find the velocity and acceleration of the intersection point $P$.
\end{problem}

\begin{problem}{1.08}
A thin rod $AB$ rests obliquely on a fixed semi-cylindrical surface of radius $R$. The end $A$, which touches the ground, moves along the horizontal plane with constant speed $v$, as shown. When the angle between the rod and the ground is $\theta$, find:
\begin{enumerate}[label=(\arabic*),leftmargin=2.2em,itemsep=1pt,topsep=2pt]
\item the velocity $v_P$ of the point $P$ of the rod that is in contact with the cylindrical surface;
\item the velocity $v_{P'}$ with which the point of intersection $P'$ of the rod and the cylindrical surface moves.
\end{enumerate}
\end{problem}
